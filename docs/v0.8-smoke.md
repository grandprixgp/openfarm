## Copy-Paste Smoke Script (`/var/tmp/of-v08-smoke`)

```bash
set -euo pipefail

OPENFARM_REPO="$HOME/projects/openfarm"
SMOKE_ROOT="/var/tmp/of-v08-smoke"
SMOKE_REPO="$SMOKE_ROOT/repo"
SMOKE_STATE="$SMOKE_ROOT/state"
ENDPOINT="https://api.commandcode.ai/provider/v1"
MODEL_ID="deepseek/deepseek-v4.1-flash"
KEY_FILE="$HOME/.config/openfarm/openfarm.key"

rm -rf "$SMOKE_ROOT"
mkdir -p "$SMOKE_REPO" "$SMOKE_STATE"
cp -R "$OPENFARM_REPO/bench/py-bounded-full/repo/." "$SMOKE_REPO/"

cd "$SMOKE_REPO"
git init -q -b main
git config user.email "smoke@local.invalid"
git config user.name "smoke"
git add -A && git commit -q -m "initial bench baseline"

PYTHONPATH="$OPENFARM_REPO" python3 -m openfarm_core.cli --repo "$SMOKE_REPO" adopt \
  --lang python \
  --state-dir "$SMOKE_STATE" \
  --key-file "$KEY_FILE" \
  --json

# Configure DeepSeek (`tier:dsf`) across workers, coordinator, and supervisor with ZDR header
python3 - <<PY
from pathlib import Path
p = Path(".openfarm/config.toml")
txt = p.read_text(encoding="utf-8")
txt = txt.replace("https://api.example.invalid/v1", "$ENDPOINT")
txt = txt.replace('"lesser-coder"', '"$MODEL_ID"')
txt = txt.replace('micromanager = "claude-acp"', 'micromanager = "off"')
txt = txt.replace('mode = "active"', 'mode = "off"')
txt = txt.replace(
    'max_tokens = 32768',
    'headers = { "x-cmd-zdr" = "1" }\nmax_tokens = 32768',
)
# Replace just the [swarm] block cleanly while preserving [harness]
before, _swarm_sep, after = txt.partition("\n[swarm]\n")
_unused, harness_sep, harness_tail = after.partition("\n[harness]\n")
updated = (
    before
    + """

[tiers.dsf]
model = "$MODEL_ID"
efforts = ["off", "low", "medium", "high", "xhigh", "max"]
effort_param = "reasoning_effort"

[swarm]
enabled = true
coordinator = "tier:dsf"
supervisor = "tier:dsf"
coordinator_effort = "max"
supervisor_effort = "max"
default_tier = "dsf"
escalate_after = 2
handbacks_per_card = 2
stall_after = "5m"
"""
    + harness_sep
    + harness_tail
)
p.write_text(updated, encoding="utf-8")
PY

cp "$OPENFARM_REPO/bench/py-bounded-full/card.md" "$SMOKE_REPO/cards/CARD-1.md"
git add -A && git commit -q -m "configure v0.8 tier:dsf smoke card"

# 1. Pre-flight check & zero-call dry run
PYTHONPATH="$OPENFARM_REPO" python3 -m openfarm_core.cli --repo "$SMOKE_REPO" check CARD-1 --baseline --json
PYTHONPATH="$OPENFARM_REPO" python3 -m openfarm_core.cli --repo "$SMOKE_REPO" run --cards 1 --dry-run --json

# 2. Live run (capped at $1.00; workers + supervisor + coordinator all on tier:dsf)
PYTHONPATH="$OPENFARM_REPO" python3 -m openfarm_core.cli --repo "$SMOKE_REPO" run --cards 1 --spend 1.00 --json

# 3. Inspect dossier, board, swarm roster, and TUI snapshot
PYTHONPATH="$OPENFARM_REPO" python3 -m openfarm_core.cli --repo "$SMOKE_REPO" dossier CARD-1
PYTHONPATH="$OPENFARM_REPO" python3 -m openfarm_core.cli --repo "$SMOKE_REPO" board read --as "*" --since 0
PYTHONPATH="$OPENFARM_REPO" python3 -m openfarm_core.cli --repo "$SMOKE_REPO" tui --once --view swarm
PYTHONPATH="$OPENFARM_REPO" python3 -m openfarm_core.cli --repo "$SMOKE_REPO" tui --once --view board

# 4. Run hidden verification tests on the produced card branch (if `in-review`)
BRANCH=$(git branch --list 'openfarm/CARD-1-*' | tr -d ' *')
if [ -n "$BRANCH" ]; then
  git checkout -q "$BRANCH"
  cp -R "$OPENFARM_REPO/bench/py-bounded-full/hidden/." "$SMOKE_REPO/"
  python3 -m unittest discover -v -s tests -t .
fi
```
